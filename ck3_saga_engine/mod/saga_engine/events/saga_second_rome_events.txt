# Second Rome -- Imperial Restoration Saga
# Saga Engine Phase 2, batch 1 (setup stage) + batch 2 (escalation stage 1)
#
# Reserved ID block: saga_second_rome.1000-1999 (see docs/SPEC.md)
# Batch 1: 1000-1003, the setup arc.
# Batch 2: 1004-1007, escalation stage 1 (rival pressure campaign). Next
# free id after this batch: 1008.
#
# 1000  The Weight of the Purple   -- reflection on the empty title
# 1001  Whispers in the Court      -- the court doubts you're truly emperor
# 1002  A Voice Denies You         -- the rival claimant is revealed
# 1003  The First Trial            -- first real legitimacy stakes; closes
#                                     the setup arc (sets
#                                     second_rome_setup_complete, which the
#                                     story_cycle's escalation-1 kickoff
#                                     effect_group watches for)
# 1004  The Rival Musters Support  -- rival campaigns for foreign non-recognition
# 1005  A Courtier's Wandering Eye -- rival's agents try to turn a courtier
# 1006  The Rival's Envoy          -- public demand to renounce the title
# 1007  Test of Arms               -- rival's allies muster at the border;
#                                     closes escalation stage 1 (sets
#                                     second_rome_escalation_1_complete for
#                                     escalation stage 2, a later batch)
##################################################

saga_second_rome.1000 = {
	type = character_event

		reference = throne_room
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		any_owned_story = {
			story_type = story_second_rome
		}
	}

	option = {
		name = saga_second_rome.1000.a
		add_prestige = minor_prestige_gain
		add_dread = minor_dread_gain
		random_owned_story = {
			type = story_second_rome
			change_variable = {
				name = second_rome_legitimacy_counter
				add = 1
			}
		}
	}

	option = {
		name = saga_second_rome.1000.b
		add_piety = minor_piety_gain
		add_character_flag = second_rome_measured_claim
	}

	after = {
		trigger_event = {
			id = saga_second_rome.1001
			days = 5
		}
	}
}
