##################################################
# Second Rome -- Imperial Restoration Saga
# Saga Engine Phase 2, batch 1 (setup stage) + batch 2 (escalation stage 1)
# + batch 3 (ending fork)
#
# Reserved ID block: saga_second_rome.1000-1999 (see docs/SPEC.md)
# Batch 1: 1000-1003, the setup arc.
# Batch 2: 1004-1007, escalation stage 1 (rival pressure campaign).
# Batch 3: 1008-1012, the ending fork (verdict prelude + four endings).
# Next free id after this batch: 1013.
#
# 1000  The Weight of the Purple   -- reflection on the empty title
# 1001  Whispers in the Court      -- the court doubts you're truly emperor
# 1002  A Voice Denies You         -- the rival claimant is revealed
# 1003  The First Trial            -- first real legitimacy stakes; closes
#                                     the setup arc (sets
#                                     second_rome_setup_complete, which the
#                                     story_cycle's escalation-1 kickoff
#                                     effect_group watches for)
# 1004  The Rival Musters Support  -- rival campaigns for foreign non-recognition
# 1005  A Courtier's Wandering Eye -- rival's agents try to turn a courtier
# 1006  The Rival's Envoy          -- public demand to renounce the title
# 1007  Test of Arms               -- rival's allies muster at the border;
#                                     closes escalation stage 1 (sets
#                                     second_rome_escalation_1_complete for
#                                     escalation stage 2, a later batch)
# 1008  The Verdict of the Age     -- prelude: the question can no longer be
#                                     deferred; one last pull on the counter,
#                                     then sets second_rome_verdict_ready for
#                                     the story_cycle's verdict effect_group
# 1009  The Acclamation            -- ENDING, counter >= 4: recognized
#                                     restorer; the coronation history accepts
# 1010  The Hollow Crown           -- ENDING, counter 0..3: the title held,
#                                     the belief never came
# 1011  Another's Chronicle        -- ENDING, counter <= -4 (or negative with
#                                     no kin to split the house): the rival
#                                     wins the story
# 1012  At Your Own Table          -- ENDING, counter -3..-1 with living adult
#                                     dynasty kin: the claim splits the house
##################################################

# ENDING -- Dynasty Schism (counter -3..-1, with living adult dynasty kin)
saga_second_rome.1012 = {
	type = character_event
	title = saga_second_rome.1012.t
	desc = saga_second_rome.1012.desc
	theme = dynasty

	override_background = {
		reference = throne_room
	}

	left_portrait = {
		character = root
		animation = anger
	}

	right_portrait = {
		character = scope:schism_kinsman
		animation = personality_bold
	}

	trigger = {
		any_owned_story = {
			story_type = story_second_rome
		}
	}

	immediate = {
		# Pick the kinsman who leads the schism; root fallback mirrors the
		# on_action's rival-candidate fallback so the scope always exists.
		if = {
			limit = {
				exists = dynasty
				dynasty = {
					any_dynasty_member = {
						NOT = { this = root }
						is_adult = yes
					}
				}
			}
			dynasty = {
				random_dynasty_member = {
					limit = {
						NOT = { this = root }
						is_adult = yes
					}
					save_scope_as = schism_kinsman
				}
			}
		}
		else = {
			save_scope_as = schism_kinsman
		}
		if = {
			limit = { exists = dynasty }
			dynasty = {
				add_dynasty_prestige = medium_dynasty_prestige_loss
			}
		}
	}

	option = {
		name = saga_second_rome.1012.a
		add_dread = medium_dread_gain
		add_stress = minor_stress_gain
	}

	option = {
		name = saga_second_rome.1012.b
		add_prestige = medium_prestige_loss
		scope:schism_kinsman = {
			add_prestige = minor_prestige_gain
		}
	}

	after = {
		add_character_flag = second_rome_ending_dynasty_schism
		random_owned_story = {
			type = story_second_rome
			end_story = yes
		}
	}
}
